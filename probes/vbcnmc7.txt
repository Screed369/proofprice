  function converterData(sEcho, json) {
    const data = { draw: sEcho, recordsTotal: 0, recordsFiltered: 0, data: [] };

    if (!json || !json.result) return data;

    data.recordsTotal = json.result.total || 0;
    data.recordsFiltered = json.result.total || 0;

    const records = JSON.stringify(json.result.records || []);

    // reemplazar sólo si columnsName está definido
    let fixedRecords = records;
    for (let clave in columnsName) {
      if (!Object.prototype.hasOwnProperty.call(columnsName, clave)) continue;
      const title = columnsName[clave];
      const esc = escapeForRegExp(title);
      const reg = new RegExp('"' + esc + '"', "g");
      fixedRecords = fixedRecords.replace(reg, '"' + clave + '"');
    }

    try {
      data.data = JSON.parse(fixedRecords);
    } catch (e) {
      console.error("Error parseando records tras replace:", e);
      data.data = json.result.records || [];
    }

    data.data = modificaTablon(data.data, json.result.fields || []);
    return data;
  }

  function modificaTablon(registros, campos) {
    if (!Array.isArray(registros) || registros.length === 0) return registros;
    if (!Array.isArray(campos) || campos.length === 0) return registros;

    for (let i = 0; i < campos.length; i++) {
      const { type, info } = campos[i];
      const notas = info && info.notes;

      if (type === "numeric" || (type === "text" && notas === "indicador")) {
        // Buscar clave real que corresponda a este campo (preferir id si existe)
        const key = campos[i].id || Object.keys(registros[0])[i];
        if (!key) continue;

        for (let j = 0; j < registros.length; j++) {
          if (Object.prototype.hasOwnProperty.call(registros[j], key)) {
            const valor = registros[j][key];
            if (valor != null && valor !== "") {
              registros[j][key] = formateaNumerico(valor);
            }
          }
        }
      }
    }

    return registros;
  }

  function formateaNumerico(valor) {
    // Convertir a número válido
    const numero = parseFloat(
      typeof valor === "string"
        ? valor.replace(/\./g, "").replace(",", ".")
        : valor
    );

    if (isNaN(numero)) return "";

    // Separar parte entera y decimal con hasta 4 decimales
    const [entera, decimal] = numero
      .toFixed(4) // fuerza 4 decimales (con ceros)
      .replace(/\.?0+$/, "") // elimina ceros sobrantes
      .split(".");

    // Añadir puntos de miles a la parte entera
    const conMiles = entera.replace(/\B(?=(\d{3})+(?!\d))/g, ".");

    return decimal ? `${conMiles},${decimal}` : conMiles;
  }

  function fnDataTablesPipeline(url, request, callback) {
    const pageSize = request.length;
    const start = request.start;
    const draw = request.draw;

    // Parámetros básicos para CKAN datastore_search
    const data = {
      resource_id: request.resource_id || '',
      limit: pageSize,
      offset: start
    };

    // -- Manejo de ordenación enviado por DataTables --
    if (Array.isArray(request.order) && request.order.length > 0) {
      const ord = request.order[0]; // primer orden (DataTables puede enviar varios)
      const colIdx = ord.column;
      const dir = ord.dir || 'asc';

      // Determinar nombre de la columna (preferir columns[colIdx].data)
      let sortField = '';
      if (request.columns && request.columns[colIdx]) {
        sortField = request.columns[colIdx].data || request.columns[colIdx].name || '';
      }

      // Si tienes un mapeo columnsName[campo] — usarlo si existe
      if (sortField && columnsName && columnsName[sortField]) {
        sortField = columnsName[sortField];
      }

      if (sortField) {
        data.sort = sortField + ' ' + dir + ', _id asc'; // añadir orden secundario por _id
      }
    }

    // -- Manejo de búsqueda (opcional) --
    if (request.search && request.search.value) {
      data.q = request.search.value;
    }

    // Llamada AJAX al endpoint CKAN
    $.ajax({
      url: url,
      type: 'GET',
      dataType: 'json',
      data: data,
      success: function (json) {
        if (json && json.result && json.result.records) {
          const recordsFormateados = modificaTablon(
            json.result.records,
            json.result.fields || []
          );
          
          callback({
            draw: draw,
            recordsTotal: json.result.total || json.result.records.length,
            recordsFiltered: json.result.total || json.result.records.length,
            data: recordsFormateados
          });
        } else {
          callback({
            draw: draw,
            recordsTotal: 0,
            recordsFiltered: 0,
            data: []
          });
        }
      },
      error: function (xhr, status, err) {
        console.error('Error al cargar datos desde CKAN', status, err);
        callback({
          draw: draw,
          recordsTotal: 0,
          recordsFiltered: 0,
          data: []
        });
      }
    });
  }
