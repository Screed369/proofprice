// read current label from input field

  return displayInput ? displayInput.textContent.trim() : 'Unknown selection';
}

// fetch and cache data
export async function getAndCacheData(id, cache, dataPathPrefix, transformFunc) {
  if (cache.has(id)) {
    return cache.get(id);
  }

  const url = `${dataPathPrefix}${id}.json`;
  const resp = await fetch(url);

  if (!resp.ok) {
    throw new Error(`Could not load data for ID ${id} from ${dataPathPrefix}. Status: ${resp.status}`);
  }

  const raw = await resp.json();
  const transformed = transformFunc(raw);
  cache.set(id, transformed);
  return transformed;
}

// load JSON-configs
export async function loadBaseConfigs(configPaths) {
  const responses = await Promise.all(configPaths.map(path => fetch(path)));
  const failed = responses.filter(resp => !resp.ok);
  if (failed.length > 0) {
    throw new Error("Failed to load one or more chart config JSON files.");
  }
  return Promise.all(responses.map(resp => resp.json()));
}

// create scaled highcharts series which are filtered by range and scaled to 100 at the beginning
export function createScaledIndexSeries(tsMap, visibilityMap, labelsMap, lang, tMin, tMax, colorPalette = null) {
  const validIndices = tsMap || {};
  const validVisibilityMap = visibilityMap || {};
  const validLabels = labelsMap?.[lang] || {};

  // check if a valid range is set
  const hasRange = tMin != null && tMax != null;

  let seriesIndex = 0;
  const createdSeries = Object.entries(validIndices)
    .map(([idxName, ts]) => {
      let filtered;

      if (hasRange) {
        // filter by range only if tMin and tMax are provided
        filtered = ts.filter(([t]) => t >= tMin && t <= tMax);
      } else {
        // if no range is set (initial load), use all data
        filtered = ts;
      }

      if (filtered.length === 0) {
        return null;
      }

      // scale data
      const baseValue = filtered[0][1];
      const scaledData = filtered.map(([t, v]) => [t, (v / baseValue) * 100]);

      // Prepare the series object
      const seriesObject = {
        name: validLabels[idxName] || idxName,
        data: scaledData,
        visible: !!validVisibilityMap[idxName]
      };

      // color assignment
      if (colorPalette && Array.isArray(colorPalette)) {
        if (seriesIndex < colorPalette.length) {
          seriesObject.color = colorPalette[seriesIndex];
        }
      }
      seriesIndex++;
      return seriesObject;
    })
    .filter(series => series !== null && series.data.length > 0);
  return createdSeries;
}

// initialize chart listeners for dynamic select inputs
export function initializeChartListener(inputId, renderFunc) {
  const inputElement = document.getElementById(inputId);
  if (!inputElement) {
    console.error(`Hidden input '${inputId}' not found. Cannot attach chart listener.`);
    return;
  }
  // we listen for the custom 'selectchange' event (defined in select.js)
  inputElement.addEventListener("selectchange", () => {
    renderFunc(inputElement.value);
  });
}

// axis snapping to the first day of the month
export function snapExtremesToMonth(e) {
  // Only snap on user-triggered events
  if (e.trigger !== 'navigator' && e.trigger !== 'rangeSelectorInput' && e.trigger !== 'rangeSelectorButton') {
    return;
  }

  const {
    min,
    max
  } = e;
  let newMin = null;
  let newMax = null;

  // Utility fn to snap a timestamp to the first of the month
  const snapTS = (ts, isMax) => {
    if (ts === null || ts === undefined) return null;

    const d = new Date(ts);
    let year = d.getUTCFullYear();
    let month = d.getUTCMonth(); // 0-indexed month
    let day = d.getUTCDate();

    // Snapping Logic
    if (isMax) {
      // Snap Max to the start of the nearest month (15th as breakpoint)
      if (day >= 15) {
        month++;
      }
    } else {
      // Always set Min to the 1st of the current month
      if (day !== 1) {
        day = 1;
      }
    }

    // we use the timestamp for the 1st of the relevant month
    return Date.UTC(year, month, 1);
  };

  // Calculate new min
  if (min !== undefined) {
    newMin = snapTS(min, false);
  }

  // Calculate new max
  if (max !== undefined) {
    newMax = snapTS(max, true);
  }

  // Apply new extremes if they are different from the original
  if ((newMin !== null && newMin !== min) || (newMax !== null && newMax !== max)) {
    e.target.setExtremes(newMin, newMax, true, false, {
      trigger: 'callback'
    });
    // Return false to cancel the original setExtremes event
    return false;
  }
}

// csv-export
export function CsvExport(chart, pidPrefix, pid) {
  // generate raw CSV string
  const raw = chart.getCSV();
  const rows = raw.split('\n');

  const processedRows = rows.map((row, rowIndex) => {
    const cells = row.split(';');

    const processedCells = cells.map((cell, cellIndex) => {
      // remove quotes
      const c = cell.replace(/^"|"$/g, '');
      if (rowIndex > 0 && cellIndex === 0) {
        const dateObj = new Date(c);

        // check if parsing was successful and format it
        if (!isNaN(dateObj)) {
          const year = dateObj.getFullYear();

          // get month (0-indexed) and pad with a leading zero
          const month = String(dateObj.getMonth() + 1).padStart(2, '0');
          return `${year}-${month}`;
        }

        // fallback if parsing fails
        return c;
      }

      // processing for numeric cells
      const normalized = c.replace(',', '.');
      const num = parseFloat(normalized);

      if (!isNaN(num)) {
        // Round and reformat with comma
        return num.toFixed(1).replace('.', ',');
      }
      return c;
    });

    // rejoin with semicolon & quote text cells
    return processedCells.map(c => `"${c}"`).join(';');
  });

  // build final csv
  const finalCsv = processedRows.join('\n');

  // Download manually
  const blob = new Blob([finalCsv], {
    type: 'text/csv;charset=utf-8;'
  });
  const link = document.createElement('a');
  link.href = URL.createObjectURL(blob);
  link.setAttribute('download', `${pidPrefix}-${pid}.csv`);
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
}
