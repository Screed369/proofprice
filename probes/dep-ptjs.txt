function Export(items, filename) {
    const replacer = (key, value) => value === null ? '' : value;
    const header = Object.keys(items[0]);
    let csv = items.map(row => header.map(fieldName => JSON.stringify(row[fieldName], replacer)).join(';'));
    csv.unshift(header.join(';'));
    csv = csv.join('\r\n');

    var link = document.createElement("a");
    link.id = "lnkDwnldLnk";
    document.body.appendChild(link);
    blob = new Blob([decodeURIComponent('%ef%bb%bf') /*prepend bom*/, csv], { type: 'text/csv;charset=utf-8' });
    var csvUrl = window.webkitURL.createObjectURL(blob);
    jQuery("#lnkDwnldLnk")
        .attr({

        });

    document.body.removeChild(link);
}

function CarregarCombo(pUrl, pData, pIdCombo, pValue, pText, pSelecione, pSelected) {
    var opt = '';
    $.ajax({
        url: pUrl,
        type: "GET",
        datatype: "json",
        data: pData,
        beforeSend: function (XMLHttpRequest) {
            $(pIdCombo).empty();
            if (pSelecione) {
                opt += '<option value="">Selecione</option>';
            }
        },
        success:
            function (data, textStatus, XMLHttpRequest) {
                if (data.status) {
                    $.each(data.resultado, function (index, item) {
                        var value = item[pValue];
                        var text = item[pText];
                        var selected = "";
                        if (value == pSelected) {
                            selected = "selected";
                        }
                        opt += '<option value="' + value + '" ' + selected + '>' + text + '</option>';
                    });
                    $(pIdCombo).append(opt);
                    var sel = $(pIdCombo);
                    var selected = sel.val(); // cache selected value, before reordering
                    var opts_list = sel.find('option');
                    opts_list.sort(function (a, b) { return $(a).text().localeCompare($(b).text()); });
                    sel.html('').append(opts_list);
                    sel.val(selected); // set cached selected value
                    $(pIdCombo + " option:first").before($(pIdCombo + '>option[value=""]'));

                    var optionValues = [];
                    $('' + pIdCombo + ' option').each(function () {
                        if ($.inArray(this.value, optionValues) > -1) {
                            $(this).remove()
                        } else {
                            optionValues.push(this.value);
                        }
                    });

                }
            },
        error:
            function (data, textStatus, XMLHttpRequest) {
                console.log("Error: " + textStatus);
            },
        complete: function () {
            

        }
    });
}

function ExportFicheiro(tipoFicheiro, items, fl_header, filename) {
    var file = "";
    switch (tipoFicheiro) {
        case 'CSV':
            const replacer = (key, value) => value === null ? '' : value;
            const header = Object.keys(items[0]);
            file = items.map(row => header.map(fieldName => JSON.stringify(row[fieldName], replacer)).join(';'));
            if (fl_header) {
                file.unshift(header.join(';'));
            }
            file = file.join('\r\n');
            break;
        case 'XML':
            file = items;
            break;
    }

    var link = document.createElement("a");
    link.id = "lnkDwnldLnk";
    document.body.appendChild(link);
    blob = new Blob([decodeURIComponent('%ef%bb%bf'), file], { type: 'text/' + tipoFicheiro + ';charset=utf-8' });
    var fileUrl = window.webkitURL.createObjectURL(blob);
    jQuery("#lnkDwnldLnk")
        .attr({
            'download': filename,
            'href': fileUrl
        });
    jQuery('#lnkDwnldLnk')[0].click();
    document.body.removeChild(link);
}
