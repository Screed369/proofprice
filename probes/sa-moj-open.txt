  import { setupNajizVueApp } from "/_catalogs/masterpage/MOJPortalDesign2026/dist/ce/vue-app.js";

  /* Language: same detection as the header (Arabic when the URL contains /ar/) */
  const IS_AR = /\/ar(\/|$)/i.test(window.location.pathname);

  const LABELS = {
    ar: {
      flagAlt: "علم المملكة العربية السعودية",
      message: "موقع حكومي مسجل لدى هيئة الحكومة الرقمية",
      howToVerify: "كيف تتحقق؟",
      tools: "أدوات تسهيل الوصول",
      displaySettings: "إعدادات الرؤية",
      normal: "الوضع العادي",
      dark: "الوضع الليلي",
      gray: "الوضع الرمادي",

      httpsTitle: "المواقع الحكومية تستخدم بروتوكول HTTPS",

    },
    en: {
      flagAlt: "Flag of Saudi Arabia",
      message: "An official website registered with the Digital Government Authority",
      howToVerify: "How to verify?",
      tools: "Accessibility tools",
      displaySettings: "Display Settings",
      normal: "Normal Mode",
      dark: "Dark Mode",
      gray: "Gray Mode",
      bright: "High brightness",
      zoomOut: "Decrease text size",
      zoomIn: "Increase text size",
      signLanguage: "Sign language support",
      panelLabel: "How to verify a government website",
      govTitle: "Official website links end with gov.sa",
      govText: "All official websites of government entities in Saudi Arabia end with gov.sa",
      httpsTitle: "Government websites use HTTPS",
      httpsText: "Secure websites in Saudi Arabia use the HTTPS protocol for encryption.",
      dgaAlt: "Digital Government Authority",
      registeredWith: "Registered with the Digital Government Authority under number:",
      loadError: "This section could not be loaded",
    },
  };
  const T = IS_AR ? LABELS.ar : LABELS.en;

  /* DGA registration (third row of the verification panel) */
  const DGA_REG_NUMBER = "20260609252";
  const DGA_LICENSE_URL =
    "https://raqmi.dga.gov.sa/platforms/platforms/98887777-b59d-443c-ab16-8222cf955bb5/platform-license";

  /* ---------- Display modes (the digital stamp is the only place they are set) ---------- */
  const VISION_STORAGE_KEY = "moj.visionMode";
  const VISION_CLASSES = { 2: "vision-dark", 3: "vision-gray", 4: "vision-bright" };

  function readStoredVision() {
    try {
      const v = Number(localStorage.getItem(VISION_STORAGE_KEY));
      return [1, 2, 3, 4].includes(v) ? v : 1;
    } catch {
      return 1;
    }
  }

  const MIN_SCALE = -2;
  const MAX_SCALE = 3;
  const SCALE_STEP = 0.0625; // 1/16 of the base size per step
  const STORAGE_KEY = "moj.a11y";
  const SIGN_LANGUAGE_URL = "https://najiz.sa/applications/video-chat";

  function readPrefs() {
    try {
      return JSON.parse(localStorage.getItem(STORAGE_KEY)) || {};
    } catch {
      return {};
    }
  }

  try {
    const prefs = readPrefs();
    const app = await setupNajizVueApp({
      template: document.getElementById("digital-stamp-tpl").innerHTML,
      data() {
        return {
          isAr: IS_AR,
          t: T,
          DGA_REG_NUMBER,
          DGA_LICENSE_URL,
          MIN_SCALE,
          MAX_SCALE,
          isVerificationOpen: false,
          textScale: Number.isInteger(prefs.textScale) ? prefs.textScale : 0,
          isVisionOpen: false,
          visionStyle: {},
          visionMode: readStoredVision(),
          visionModes: [
            { value: 1, key: "normal", label: T.normal },
            { value: 2, key: "dark", label: T.dark },
            { value: 3, key: "gray", label: T.gray },
            { value: 4, key: "bright", label: T.bright },
          ],
          // Absolute path: an inline module's import.meta.url is the *page* URL,
          // so "../assets/..." resolved relative to the SharePoint page, not the dist folder.
          flagIcon: "/_catalogs/masterpage/MOJPortalDesign2026/assets/saudi-flag-mark.svg",
        };
      },
      computed: {
        // Falls back to the built-in text if the LABELS key is missing in a merged copy.
        registeredLabel() {
          return (
            this.t.registeredWith ||
            (this.isAr
              ? "مسجل لدى هيئة الحكومة الرقمية برقم:"
              : "Registered with the Digital Government Authority under number:")
          );
        },
      },
      watch: {
        visionMode(mode) {
          this.applyVision(mode);
        },
      },
      mounted() {
        this.applyPrefs();
        this.applyVision(this.visionMode);
        document.addEventListener("click", this.onDocumentClick);
        document.addEventListener("keydown", this.onKeydown);
        window.addEventListener("resize", this.positionVision);
        window.addEventListener("scroll", this.positionVision, true);
      },
      beforeUnmount() {
        document.removeEventListener("click", this.onDocumentClick);
        document.removeEventListener("keydown", this.onKeydown);
        window.removeEventListener("resize", this.positionVision);
        window.removeEventListener("scroll", this.positionVision, true);
      },
      // These methods were missing, so every @click did nothing.
      methods: {
        toggleVerification() {
          this.isVerificationOpen = !this.isVerificationOpen;
        },
        toggleVision() {
          this.isVisionOpen = !this.isVisionOpen;
          if (this.isVisionOpen) {
            this.$nextTick(() => {
              this.positionVision();
              // move focus to the selected option for keyboard users
              document.getElementById("ds-vision-" + this.visionMode)?.focus();
            });
          }
        },
        /* Place the menu right under the button, kept inside the window.
           Arabic: menu starts at the button's left edge; English: ends at its right edge. */
        positionVision() {
          if (!this.isVisionOpen) return;
          const btn = this.$refs.visionBtn;
          const menu = this.$refs.visionMenu;
          if (!btn || !menu) return;
          const r = btn.getBoundingClientRect();
          const w = menu.offsetWidth || 192;
          const gap = 6;
          let left = this.isAr ? r.left : r.right - w;
          left = Math.max(8, Math.min(left, window.innerWidth - w - 8));
          this.visionStyle = { top: `${Math.round(r.bottom + gap)}px`, left: `${Math.round(left)}px` };
        },
        closeVision(returnFocus = false) {
          if (!this.isVisionOpen) return;
          this.isVisionOpen = false;
          if (returnFocus) this.$refs.visionBtn?.focus();
        },
        onDocumentClick() {
          this.closeVision();
        },
        onKeydown(event) {
          if (event.key === "Escape") this.closeVision(true);
        },
        applyVision(mode) {
          const root = document.documentElement;
          Object.values(VISION_CLASSES).forEach((c) => root.classList.remove(c));
          if (VISION_CLASSES[mode]) root.classList.add(VISION_CLASSES[mode]);
          try {
            localStorage.setItem(VISION_STORAGE_KEY, String(mode));
          } catch {
            /* storage unavailable – mode still applies for this page view */
          }
        },
        changeScale(delta) {
          this.textScale = Math.min(MAX_SCALE, Math.max(MIN_SCALE, this.textScale + delta));
          this.applyPrefs();
        },
        openSignLanguage() {
          window.open(SIGN_LANGUAGE_URL, "_blank", "noopener");
        },
        applyPrefs() {
          const root = document.documentElement;
          root.classList.remove("ds-high-contrast"); // old contrast toggle, replaced by the display modes
          root.style.fontSize = this.textScale === 0 ? "" : `${100 + this.textScale * SCALE_STEP * 100}%`;
          try {
            localStorage.setItem(
              STORAGE_KEY,
              JSON.stringify({ textScale: this.textScale }),
            );
          } catch {
            /* storage unavailable – settings apply for this page view only */
          }
        },
      },
    });

    app.mount("#digital-stamp-root");
  } catch (error) {
    console.error("[digital-stamp]", error);
    document.getElementById("digital-stamp-root").textContent = T.loadError;
  }
</script>
