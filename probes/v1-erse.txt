<script>
    (function () {
        function initializeNavigation() {
            var navigation = document.getElementById('navbarSupportedContent');
            var toggle = document.querySelector('#main-nav .navbar-toggler');
            var menu = navigation ? navigation.querySelector('.navbar-nav') : null;

            if (!navigation || !toggle || !menu) {
                return;
            }

            function isMobileNavigation() {
                return window.innerWidth < 992;
            }

            function isNavigationOpen() {
                return navigation.classList.contains('show') || navigation.classList.contains('collapsing');
            }

            function syncNavigationState() {
                var isOpen = isMobileNavigation() && isNavigationOpen();
                document.body.classList.toggle('navigation-open', isOpen);
                toggle.classList.toggle('is-open', isOpen);

                if (!navigation.classList.contains('collapsing')) {
                    toggle.setAttribute('aria-expanded', isOpen ? 'true' : 'false');
                }
            }

            function closeNavigation(restoreFocus) {
                if (!isMobileNavigation() || !isNavigationOpen()) {
                    return;
                }

                toggle.click();

                if (restoreFocus) {
                    window.setTimeout(function () {
                        toggle.focus();
                    }, 0);
                }
            }

            function resetDesktopState() {
                if (isMobileNavigation()) {
                    syncNavigationState();
                    return;
                }

                navigation.classList.remove('show', 'collapsing');
                navigation.style.height = '';
                document.body.classList.remove('navigation-open');
                toggle.classList.remove('is-open');
                toggle.setAttribute('aria-expanded', 'false');

                navigation.querySelectorAll('.dropdown.show').forEach(function (dropdown) {
                    dropdown.classList.remove('show');
                    var dropdownToggle = dropdown.querySelector('.dropdown-toggle');
                    var dropdownMenu = dropdown.querySelector('.dropdown-menu');

                    if (dropdownToggle) {
                        dropdownToggle.setAttribute('aria-expanded', 'false');
                    }

                    if (dropdownMenu) {
                        dropdownMenu.classList.remove('show');
                    }
                });
            }

            new MutationObserver(syncNavigationState).observe(navigation, {
                attributes: true,
                attributeFilter: ['class']
            });

            document.addEventListener('click', function (event) {
                if (!isMobileNavigation() || !isNavigationOpen()) {
                    return;
                }

                if (menu.contains(event.target) || toggle.contains(event.target)) {
                    return;
                }

                closeNavigation(false);
            });

            document.addEventListener('keydown', function (event) {
                if (event.key === 'Escape') {
                    closeNavigation(true);
                }
            });

            window.addEventListener('resize', resetDesktopState);
            toggle.addEventListener('click', function () {
                window.requestAnimationFrame(syncNavigationState);
            });

            resetDesktopState();
        }

        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', initializeNavigation);
        } else {
            initializeNavigation();
        }
    })();
</script>
