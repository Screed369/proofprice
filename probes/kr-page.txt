function fn_popLoop(res){

	for (var i = 0; i < res.popList.length; i++) {
		var url = res.popList[i].fileUrl;
		var width = res.popList[i].popupWidthSize;
		var height = res.popList[i].popupVrticlSize;
		var top = res.popList[i].popupVrticlLc;
		var left = res.popList[i].popupWidthLc;
		var popupId = res.popList[i].popupId;
		var stopvewSetupAt = res.popList[i].stopvewSetupAt;
		
		url = url + "?popupId="+popupId;
		url = url + "&stopvewSetupAt="+stopvewSetupAt;
		
		var cookieCheck = "";
		var data = {
				popupId:popupId	
		};
		
		$.ajax({
			type : "POST", 
			url : "/pt/main/getPopupCookie.do",			
			dataType:"json",
			data : data,
			traditional : true, 
			success : function (res) {
				// empty
				cookieCheck = res.cookieStr;
				//console.log("cookieCheck:" + cookieCheck);
			}, 
			error: function(request, status, error, response){
		    	//alert("장애가 발생했습니다. 관리자에게 문의 하세요.");
				cookieCheck = "";
		    },
		    async:false
		});
		
		if (cookieCheck != "Y") {
			var openWindows = window.open(  url,
											popupId,
											"width="+ width
											+ ",height="+ height
											+ ",top="+ top
											+ ",left="+ left
											+ ",toolbar=no,status=no,location=no,scrollbars=yes,menubar=no,resizable=yes"); 
		}		
	}
}

function fn_getBbs() {
			
	var bid 		= $("input[name=bbsId]").val();
	var divSwiper 	= "";
	
	divSwiper += "<div class='notice-head'>";
	divSwiper += "<span class='notice'>&nbsp;</span>";
	divSwiper += "<span class='date'>&nbsp;</span>";
	divSwiper += "</div>";		
	divSwiper += "<p class='notice-title'>&nbsp;</p>";
	divSwiper += "<p class='notice-body'><span class='body-title'>&nbsp;</span></p>";
	
	$("#bbsTxt0").html(divSwiper);
	$("#bbsTxt1").html(divSwiper);
	$("#bbsTxt2").html(divSwiper);
	$("#bbsTxt3").html(divSwiper);
	$("#bbsTxt4").html(divSwiper);
	
	var data = {bbsId : bid};
	
	$.ajax({
		type : "POST", 
		url : "/pt/main/bbsPreview.do",
		data : data,
		dataType:"json",
		traditional : true, 
		success : function (res) {
			
			var nttSj 				= ""; // 제목
			var nttCn 				= ""; // 내용
			var answerCn 			= ""; // 답변
			var frstRegisterPnttm 	= ""; // 날짜
			var nttId 				= ""; // ID
			
			//console.log(">>>"+res.resultList.length);
			
			for (var i=0; i<res.resultList.length; i++) {
				
				// 제목
				nttSj 		= res.resultList[i].nttSj;
				jsonString 	= JSON.stringify(res.resultList[i].nttSj);
				jsonString 	= jsonString.replace(/&quot;/g,'"');
				nttSj 		= jsonString.replace(/^"(.*)"$/,'$1');
				nttSj 		= nttSj.length > 25 ? nttSj.substr(0,25)+"..." : nttSj;
				
				// 내용
				var jsonString 	= JSON.stringify(res.resultList[i].nttCn);
				jsonString 		= jsonString.replace(/&quot;/g,'"');
				nttCnTmp1 		= jsonString.replace(/^"(.*)"$/,'$1');
				nttCn 			= nttCnTmp1.replace(/(?:\\r\\n|\\r|\\n)/g,'<br />'); 


				//날짜
				frstRegisterPnttm 	= res.resultList[i].frstRegisterPnttm;
				nttId 				= res.resultList[i].nttId;
				bbsNm 				= res.boardMasterVO.bbsNm;
				
				divSwiper = "";
				divSwiper += "<div class='notice-head'>";
				divSwiper += "<span class='notice'>"+bbsNm+"</span>";
				divSwiper += "<span class='date'>"+frstRegisterPnttm+"</span>";
				divSwiper += "</div>";
				divSwiper += "<a href='javascript:fn_getBbsDtl("+nttId+");' class='link' title='제목 : "+nttSj+" 보기'>";
				divSwiper += "<p class='notice-title'>"+nttSj+"</p>";
				divSwiper += "<p class='notice-body'><span class='body-title'>"+nttCn+"</span></p>";
				divSwiper += "</a>";
				
				$("#bbsTxt"+i).html(divSwiper);
									
			}				
		}, 
		error: function(request, status, error, response){
	    	alert("장애가 발생했습니다. 관리자에게 문의 하세요.");
	    },
	    async:false
	});
}

function fn_getBbsDtl(nttId) {
	var bbsId = $("input[name=bbsId]").val();
	if(bbsId=='bbsFaq'){
		location.href="/pt/bbs/faqDetail.do?faqId="+nttId+"&mobileAt=";
	}else{
		location.href="/pt/bbs/bbsDetail.do?nttId="+nttId+"&bbsId="+bbsId+"&mobileAt=";	
	}
}
