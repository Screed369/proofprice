        var dateStartupFunctions = function dateStartup() {

            {

                $('.date').datepicker({
                    changeMonth: true,
                    changeYear: true,
                    yearRange: 'c-50:c+10',
                    onSelect: function (dateText, inst) { $(this).valid(); if (inst.input) inst.input.trigger('change'); }
                });

                //Filtro de datas em range
                $("div.dateRange input[type=text].date").on('change', function () {
                    var option = $(this).parent().hasClass('dateRangeStart') ? "minDate" : "maxDate";
                    var date = $(this).val();
                    if (date) {
                        $(this).parents(".DateRangeParent, .container").first().find(".date").not(this).datepicker("option", option, date);
                    }
                });
                //Trigger logo ao entrar na página, para os campos preenchidos em servidor
                $("div.dateRange input[type=text].date").trigger('change');

                //Restrição para as datas nunca serem inferiores a 01-01.1800
                $(".date").each(function () {
                    if ($(this).datepicker("option", 'minDate') == null)
                        $(this).datepicker("option", 'minDate', '01-01-1800')
                });

                //Restrição minDate
                $("input.date[data-val-startdate-mindate]").each(function () {
                    $(this).datepicker("option", "minDate", $(this).attr('data-val-startdate-mindate'));
                });
                $("input.date[data-val-startdatebiggerthan]").each(function () {
                    $(this).datepicker("option", "minDate", getValueByFieldID('#' + $(this).attr('data-val-startdatebiggerthan-datebiggerthan')));

                //Restrição maxDate
                $("input.date[data-val-enddate-maxdate]").each(function () {
                    $(this).datepicker("option", "maxDate", $(this).attr('data-val-enddate-maxdate'));
                });
                $("input.date[data-val-enddatesmallerthan]").each(function () {
                    $(this).datepicker("option", "maxDate", getValueByFieldID('#' + $(this).attr('data-val-enddatesmallerthan-datesmallerthan')));
                });
            }
        }

        var ModalPopupChoosenFunction = function ModalPopupChoosen(itemID, designacao) {
            var currentModalSearch = $("#currentModalSearch").val();
            var currentModalMultiSearch = $("#currentModalMultiSearch").val();
            if (currentModalSearch.length > 0)
            {
                if (!$("#" + currentModalSearch + " option[value='" + itemID + "']").length > 0)
                    $("#" + currentModalSearch).append("<option value='" + itemID + "'>" + designacao + "</option>")
                $("#" + currentModalSearch).val(itemID);
                $("#" + currentModalSearch).parent().find("input[type='text']:first").val(designacao);
                $("#" + currentModalSearch).trigger('change');
            }
            else
            {
                $.ajax({
                    url: '/Home/GetDropDownMultiSearchListNewRow',
                    cache: false,
                    data: { ID: itemID, Designacao: designacao },
                    success: function (html) {
                        var prefixo = $("#" + currentModalMultiSearch + " input[id$='Prefixo']").val();
                        
                        //adiciona a linha à lista
                        $("#" + currentModalMultiSearch + " div.DropDownMultiSearchList_EditorRows").append(html); 
                        $("#" + currentModalMultiSearch + " div.DropDownMultiSearchList_EditorRows .FakeDropDownClose:last button").button({ icons: { primary: "ui-icon-close" }, text: false });

                        //Adiciona o prefixo correcto aos campos
                        $("#" + currentModalMultiSearch + " div.DropDownMultiSearchList_EditorRows div.row:last input[type=hidden]").each(function(){
                            //alert($(this).attr('name'));
                            $(this).attr('name', prefixo + '.' + $(this).attr('name'));
                            $(this).attr('id', prefixo + '_' + $(this).attr('name'));
                        });
                    }
                });
            }
            $("#modalDiv").dialog("close");
        }



        // ####### FUNÇÕES AUXILIARES  ##########
        function baseUrl() {
            var hasVD = false;
            var href = window.location.href.split('/');

            if (hasVD)
                return href[0] + '//' + href[2] + '/' + href[3] + '/';
            else
                return href[0] + '//' + href[2] + '/';
        }

        function getValueByFieldID(id) {
            var control = $(id);
            var controltype = control.attr('type');
            var actualvalue = "";

            switch (controltype) {
                case 'checkbox':
                    actualvalue = control.attr('checked').toString(); break;
                case 'select':
                    actualvalue = $('option:selected', control).val(); break;
                default:
                    actualvalue = control.val(); break;
            }
            return actualvalue;
        }

        function removeAll(value, char) {
            while (value.indexOf(char) != -1) {
                value = value.replace(char, '');
            }
            return value;
        }
