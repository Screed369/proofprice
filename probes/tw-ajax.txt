<script type="text/javascript">
var downLoadType = 1;
var baseUrl ="https://plvr.land.moi.gov.tw/";  
 
	 
$(".advDownloadClass").hide();
function chgDownloadType(obj){
	downLoadType = obj.value;
	if (downLoadType == 2){
		$(".advDownloadClass").show();
	} else{ 
		$(".advDownloadClass").hide();
	}
}

function preDownload(){
	if (downLoadType==2){ 
		startOpenData();
	}else{ 
		var fileFormat = $("#fileFormatId").val();
		var url = "/Download?type=zip&fileName=lvr_land"+fileFormat+".zip";
		$("#download-button").attr("href",url);
		var href = baseUrl + $('#download-button').attr('href');
		$("#download-button").click();
		window.location.href = href;   
	}
}

function init(){
	$("#totalMsg").html("");
	$("#nowMsg").html("");
	$("#mainMsg").attr("style","display:none;");
}

function startOpenData(){
	init();
	fs = new zip.fs.FS();
	
	var fileFormat = $("#fileFormatId").val();
	var cnt = $(".checkBoxGrp" ).filter(':checked').length; 
	if (cnt < 1){ 
		alert("請選擇所要下載的區域及交易類別");
		$("#div_body").unmask();
		return false;
	}else{
		if (cnt > 6 ){ 
			alert("您選擇了檔案數共:" + cnt + "個，已超過設定限制6個，可以選「全國資料」再自行篩選");
			$("#div_body").unmask();
			return false;
		}
			var fOpt = {
					"msg" : "您選擇了檔案數共:" + cnt + "個 ，下載中，請稍候...",
					"overlay" : true,
					"zindex" : 99999
			};
			qt.util.showLoading(fOpt);
//		$("#div_body").mask("您選擇了檔案數共:" + cnt + "個 ，下載中，請稍候...");
	} 
	//fs.root.addHttpContent("readme",  baseUrl +"/Download?fileName="+"readme");
	var processCnt= 0; 
	var q = []; 
	$( ".checkBoxGrp" ).each(function( index ) {
		 var writeFileName = ($(this).val()+"."+fileFormat).toLowerCase();
		 var code = $(this).attr("code").toLowerCase()  ;
		 
		 if($(this).is(':checked')){
			 if (code !="" && code != undefined ){
				q.push(code); 
			 }
			processCnt++;
			fs.root.addHttpContent(writeFileName, baseUrl +"/Download?fileName="+writeFileName);  
// 			console.log(" fullurl:" + baseUrl +"/Download?fileName="+writeFileName);  
		 }  
	});   

	fs.root.addHttpContent("schema-land.csv", 
			baseUrl +"/Download?fileName="+"schema-land.csv"+"&q="+q.join(",")+"&type="+fileFormat);

	fs.root.addHttpContent("schema-main.csv", 
			baseUrl +"/Download?fileName="+"schema-main.csv"+"&q="+q.join(",")+"&type="+fileFormat);

	fs.root.addHttpContent("schema-park.csv", 
			baseUrl +"/Download?fileName="+"schema-park.csv"+"&q="+q.join(",")+"&type="+fileFormat);

	fs.root.addHttpContent("schema-build.csv", 
			baseUrl +"/Download?fileName="+"schema-build.csv"+"&q="+q.join(",")+"&type="+fileFormat);

	fs.root.addHttpContent("manifest.csv", 
			baseUrl +"/Download?fileName="+"manifest.csv"+"&q="+q.join(",")+"&type="+fileFormat);
 
	startDownload();  
}
</script>
