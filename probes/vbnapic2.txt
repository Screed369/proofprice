<script>
    function openDialog() {
        var el = document.getElementById("w3c-dialog");
        if(el.getAttribute("data-display") == "show") {
            el.setAttribute('data-display', "hidden");
            el.style.display = "none";
        } else {
            el.setAttribute("data-display", "show");
            el.style.display = "block";
        }
    }

    function closeDialog() {
        document.getElementById("w3c-dialog").style.display = "none";
    }

    
    function checkClass(el, val) {
        const pageOptions = ['invert', 'dark-contrast', 'low-saturation', 'high-saturation', 'monochrome'];
        
        if(pageOptions.includes(val)) {
            document.body.classList.remove('invert', 'dark-contrast', 'low-saturation', 'high-saturation', 'monochrome');
            if(el.classList.contains('active')) {
                
                el.classList.remove('active');
                document.body.classList.remove(val);
                localStorage.removeItem('bodyClass');
            } else {
                document.querySelectorAll('button').forEach(btn => btn.classList.remove('active'));
                el.classList.add('active');
                document.body.classList.add(val);
                localStorage.setItem('bodyClass', val);
            }
        }else if(el.id == 'highlight-heading') {
            if(el.classList.contains('active')) {
                el.classList.remove('active')
                document.querySelectorAll('h1, h2, h3, h5, h5, h6').forEach(e => e.classList.remove(val));
                localStorage.removeItem('highlight-heading');
            } else {
                el.classList.add('active')
                document.querySelectorAll('h1,h2,h3,h4,h5,h6').forEach(e => e.classList.add(val));
                localStorage.setItem('highlight-heading', 1);
            }
        } else if(el.id == 'highlight-link') {
            if(el.classList.contains('active')) {
                el.classList.remove('active')
                document.querySelectorAll('a').forEach(e => e.classList.remove(val));
                localStorage.removeItem(el.id);
            } else {
                el.classList.add('active')
                document.querySelectorAll('a').forEach(e => e.classList.add(val));
                localStorage.setItem(el.id, 1);
            }
        }

    }
