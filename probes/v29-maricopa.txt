            const originalAppendChild = Node.prototype.appendChild;

            const originalRemoveChild = Node.prototype.removeChild;
            const originalReplaceChild = Node.prototype.replaceChild;

            function isWatsonSidebarList(node) {
                if (!node || node.nodeType !== Node.ELEMENT_NODE || node.tagName?.toLowerCase() !== 'ul') return false;
                return node.classList?.contains('cds--side-nav__items') ||
                    /^contained-list-.*-header$/.test(node.getAttribute('aria-labelledby') || '');
            }

            function needsListItemWrapper(parent, child) {
                if (!isWatsonSidebarList(parent)) return false;
                if (!child || child.nodeType !== Node.ELEMENT_NODE) return false;
                const tagName = child.tagName?.toLowerCase();
                return tagName && tagName !== 'li' && tagName !== 'script' && tagName !== 'template';
            }

            function createWrapperFor(child) {
                const li = document.createElement('li');
                li.className = 'mari-watson-listitem-normalizer';
                li.setAttribute('data-mari-listitem-normalizer', 'true');
                li.appendChild(child);
                return li;
            }

            function getNormalizedReference(parent, referenceNode) {
                if (!isWatsonSidebarList(parent) || !referenceNode) return referenceNode;
                if (referenceNode.parentNode === parent) return referenceNode;
                const wrapper = referenceNode.closest?.('li[data-mari-listitem-normalizer="true"]');
                return wrapper?.parentNode === parent ? wrapper : referenceNode;
            }

            Node.prototype.appendChild = function(child) {
                if (needsListItemWrapper(this, child)) {
                    return originalAppendChild.call(this, createWrapperFor(child));
                }
                return originalAppendChild.call(this, child);
            };

            Node.prototype.insertBefore = function(newNode, referenceNode) {
                const normalizedReference = getNormalizedReference(this, referenceNode);
                if (needsListItemWrapper(this, newNode)) {
                    return originalInsertBefore.call(this, createWrapperFor(newNode), normalizedReference);
                }
                return originalInsertBefore.call(this, newNode, normalizedReference);
            };

            Node.prototype.replaceChild = function(newChild, oldChild) {
                const normalizedOldChild = getNormalizedReference(this, oldChild);
                if (needsListItemWrapper(this, newChild)) {
                    return originalReplaceChild.call(this, createWrapperFor(newChild), normalizedOldChild);
                }
                return originalReplaceChild.call(this, newChild, normalizedOldChild);
            };

            Node.prototype.removeChild = function(child) {
                if (isWatsonSidebarList(this) && child?.parentNode !== this) {
                    const wrapper = child.closest?.('li[data-mari-listitem-normalizer="true"]');
                    if (wrapper?.parentNode === this) {
                        return originalRemoveChild.call(this, wrapper);
                    }
                }
                return originalRemoveChild.call(this, child);
            };
        }

        function containsError(text) {
            if (!text) return false;
            const errorPatterns = [
                /Error in plugin tool execution/i,
                /failed to execute function/i,
                /connection refused/i,
                /dial tcp/i,
                /^Error:/i,
                /Traceback/i
            ];
            return errorPatterns.some(pattern => pattern.test(text));
        }

            instance.on('pre:send', preSendHandler);
            instance.on('pre:receive', preReceiveHandler);
            instance.once('pre:receive', disclaimerHandler);
            instance.on('receive', receiveHandler);
            instance.on('feedback', feedbackHandler);
            instance.on('restartConversation', restartConversationHandler);

            startWatsonA11yPatches();

            // Add aria-label to chat button for accessibility
            setTimeout(() => {
                const chatButton = document.querySelector('button.chatButton');
                if (chatButton && !chatButton.getAttribute('aria-label')) {
                    chatButton.setAttribute('aria-label', 'Open chat');
                }
            }, 500);

            // Add privacy policy link to chat widget footer
            setTimeout(() => {
                const chatWindow = document.querySelector('.wxo-float');
                if (chatWindow && !document.getElementById('privacy-policy-footer')) {
                    const footer = document.createElement('div');
                    footer.id = 'privacy-policy-footer';
                    footer.style.cssText = 'padding: 8px 16px; text-align: center; font-size: 0.75rem; color: #666; border-top: 1px solid #e0e0e0; background-color: #f5f5f5;';
                    footer.innerHTML = '<a href="https://www.maricopa.gov/6576/Virtual-Agent-Privacy-Policy" target="_blank" rel="noopener noreferrer" style="color: #F26724; text-decoration: none;">Privacy Policy</a>';
                    chatWindow.appendChild(footer);
                }
            }, 1000);

            // Check for agent_query URL parameter
            const urlParams = new URLSearchParams(window.location.search);
            const agentQuery = urlParams.get('agent_query');

            if (agentQuery) {
                // Wait for chat to be fully loaded before interacting
                setTimeout(() => {
                    // Find and click the chat button to open the chat
                    const chatButton = document.querySelector('button.chatButton');
                    if (chatButton) {
                        chatButton.click();

                        // Wait for chat to open, then fill in the query
                        setTimeout(() => {
                            const inputArea = document.querySelector('div.WxOChatInputArea');
                            if (inputArea) {
                                // Find the contenteditable div (TipTap/ProseMirror editor)
                                const editableDiv = inputArea.querySelector('div[contenteditable="true"]');
                                if (editableDiv) {
                                    const decodedQuery = decodeURIComponent(agentQuery);

                                    // Set the content in the contenteditable div
                                    const paragraph = editableDiv.querySelector('p');
                                    if (paragraph) {
                                        paragraph.textContent = decodedQuery;
                                    } else {
                                        editableDiv.innerHTML = `<p>${decodedQuery}</p>`;
                                    }

                                    // Trigger input events to notify the editor
                                    editableDiv.dispatchEvent(new Event('input', { bubbles: true }));
                                    editableDiv.dispatchEvent(new Event('change', { bubbles: true }));

                                    // Focus on the editor
                                    editableDiv.focus();

                                    // Move cursor to end
                                    const range = document.createRange();
                                    const sel = window.getSelection();
                                    range.selectNodeContents(editableDiv);
                                    range.collapse(false);
                                    sel.removeAllRanges();
                                    sel.addRange(range);
                                }
                            }
                        }, 500);
                    }
                }, 1000);
            }
        }
