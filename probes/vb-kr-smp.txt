		function LPAD(s, c, n) {
		    if (! s || ! c || s.length >= n) {
		        return s;
		    }

		    var max = (n - s.length)/c.length;
		    for (var i = 0; i < max; i++) {
		        s = c + s;
		    }

		    return s;
		}

		    var max = (n - s.length)/c.length;

		        s += c;

        function moneyFormmat(str) {
            var rstr = "", sign = "";

            if(str.substr(0, 1) == "-" ){
              sign = "-";
              str = str.substr(1);
            }
            var arr = str.split(".");
            for(var ii = 0 ; ii < arr[0].length ; ii++){
              if( ii % 3 == 0 && ii > 0){
                rstr = arr[0].substring(arr[0].length-ii,arr[0].length-ii-1) + "," + rstr;
              }else{
                rstr = arr[0].substring(arr[0].length-ii,arr[0].length-ii-1) + rstr;
              }
            }
            rstr = sign + rstr;
            return arr.length > 1 ? rstr + "." + arr[1] : rstr;

		function postObj(obj){
		    try {

		        // 타겟 document 객체
		        var targetDocument = null;

		        // 현제페이지기준 프레임 타겟       ITARGET
		        if(obj["ITARGET"] != undefined){

		            targetDocument = window.frames[obj["ITARGET"]].document;
		        }

		        // 탑페이지기준 프레임 타겟        TARGET
		        else if(obj["TARGET"] != undefined){

		            var target = obj["TARGET"].split(".");

		            targetDocument = top;

		            if(target.length>0){
		                if(target[0] != ""){
		                    for(var idx=0; idx<target.length; idx++){

		                        // top 문자 처리
		                        if(target[idx] == "top"){
		                            targetDocument = targetDocument.top;
		                        }

		                        // opener 문자 처리
		                        else if(target[idx] == "opener"){
		                            targetDocument = targetDocument.opener;
		                        }

		                        // 프레임명 문자 처리
		                        else{
		                            targetDocument = targetDocument.frames[target[idx]];
		                        }
		                    }
		                }
		            }

		        }else{
		            targetDocument = document;
		        }

		        // 타겟 검사
		        if(targetDocument == null) {
		            alert("타겟지정이 잘못되었습니다. 다시지정하여주세요.");
		            return false;
		        }

		        // 폼생성
		        var fm = targetDocument.createElement("form");

		        // 폼설정
		        fm.method="post";
		        fm.action=obj["URL"];

		        if(obj["ITARGET"] != undefined) {
		            fm.target=obj["ITARGET"];
		        }
		        if(obj["TOTARGET"] != undefined){
		            fm.target=obj["TOTARGET"];
		        }

		        // 오브젝트변수를 form 에 적용
		        for(i in obj){
		            if(i != "TARGET" && i != "URL" && i != "TOTARGET" && i != "ITARGET"){

		                if(typeof obj[i] == "object"){
		                    var obj2 = obj[i];
		                    if(obj2 == null){
		                        var input=targetDocument.createElement("input");
		                        input.setAttribute ("name", i);
		                        input.setAttribute ("type", "hidden");
		                        input.setAttribute ("value", obj[i]);
		                        fm.appendChild(input);;
		                    }else{
		                        for(j in obj2){
		                            if(typeof obj2[j] != "function"){
		                                var input=targetDocument.createElement("input");
		                                input.setAttribute ("name", i);

		                                input.setAttribute ("type", "hidden");
		                                input.setAttribute ("value", obj2[j]);
		                                fm.appendChild(input);
		                            }
		                        }
		                    }
		                }else{
		                    var input=targetDocument.createElement("input");
		                    input.setAttribute ("name", i);
		                    input.setAttribute ("type", "hidden");
		                    input.setAttribute ("value", obj[i]);
		                    fm.appendChild(input);
		                }
		            }
		        }

		        // 폼추가(타겟 문서에 폼을 추가한다.)
		        targetDocument.body.appendChild(fm);

		        // 타겟 문서 캐릭터셋 저장 (복구용)
