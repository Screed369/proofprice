    function DisabledDates(jumatOnly) {
        if(jumatOnly){
            $("#dboDate").dxDateBox({
                disabledDates: function (args) {
                    const dayOfWeek = args.date.getDay();
                    const notFriday = args.view === "month" && dayOfWeek !== 5
                    return notFriday;
                }
            });
            //get jumat terakhir
            const t = new Date().getDate() + (6 - new Date().getDay() - 1) - 7 ;
            const lastJumat = new Date();
            lastJumat.setDate(t);
            $("#dboDate").dxDateBox("instance").option("value", lastJumat); //set jadi jumat terakhir
        }
        else{
            $("#dboDate").dxDateBox({
                disabledDates: function (args) {
                    return null;
                }
            });
            $("#dboDate").dxDateBox("instance").option("value", new Date()); //set jadi tgl skrg
        }
    }
