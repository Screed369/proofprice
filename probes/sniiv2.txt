                    tbl = $('#tbl_di');
                    break;

                    anio = $('#ddl_anio_verificacion').val();
                    dv = $('#dv_verificacion');
                    tbl = $('#tbl_verificacion');
                    break;
                case _produccion:
                    anio = $('#ddl_anio_produccion').val();
                    dv = $('#dv_produccion');
                    tbl = $('#tbl_produccion');
                    break;
                default:
                    anio = $('#ddl_anio_cnbv').val();
                    dv = $('#dv_cnbv');
                    tbl = $('#tbl_cnbv');
                    break;
            }
            dv.hide();
            $(".loader").show();
            $.getJSON(path + 'ReporteAPI/GetDocumentoAnio/' + tipo + '/' + anio + '/' + formato, function (json) {
                var archivo = 'https://sniiv-cors.herokuapp.com/' + json.documento;
                $.ajax({
                    url: archivo,
                    dataType: 'text',
                }).done(function (data, textStatus, jqXHR) {
                    var allRows = data.split(/\r?\n|\r/);
                    var table = '<table class="table table-striped table-condensed">';
                    for (var singleRow = 0; singleRow < allRows.length; singleRow++) {
                        if (singleRow === 0) {
                            table += '<thead>';
                            table += '<tr>';
                        } else
                            table += '<tr>';
                        var rowCells = allRows[singleRow].split(',');
                        for (var rowCell = 0; rowCell < rowCells.length; rowCell++) {
                            if (singleRow === 0) {
                                table += '<th>';
                                table += rowCells[rowCell];
                                table += '</th>';
                            } else {
                                table += '<td>';
                                table += rowCells[rowCell];
                                table += '</td>';
                            }
                        }
                        if (singleRow === 0) {
                            table += '</tr>';
                            table += '</thead>';
                            table += '<tbody>';
                        } else
                            table += '</tr>';
                        if (singleRow === lim_rows)
                            break;
                    }
                    table += '</tbody>';
                    table += '</table>';

                    dv.show();
                    tbl.empty();
                    tbl.append(table);

                    $('html, body').animate({
                        scrollTop: dv.offset().top - 150
                    }, 500);
                })
                    .fail(function (jqXHR, textStatus, errorThrown) { console.log(errorThrown); })
                    .always(function () {
                        $(".loader").hide();
                    });
            });
        }

    <h2>Financiamientos</h2>
    <p style="font-size:15px">La API de Financiamientos proporciona informaci&#xF3;n detallada sobre las acciones de financiamiento en materia de vivienda otorgadas en M&#xE9;xico. &#xA; Esta permite consultar datos desagregados por a&#xF1;o, entidad federativa, municipio y variables que describen tanto las caracter&#xED;sticas del financiamiento como el perfil sociodemogr&#xE1;fico.</p>
    <p style="font-size:15px">La informaci&#xF3;n incluye el tipo de cr&#xE9;dito, el destino del financiamiento (como adquisici&#xF3;n, mejoramiento, arrendamiento, entre otros), el grupo y nombre del organismo que otorg&#xF3; el cr&#xE9;dito, as&#xED; como variables relacionadas, como su g&#xE9;nero, rango de edad, nivel de ingresos (en UMA) y si reside en un municipio con poblaci&#xF3;n ind&#xED;gena.</p>
