    <div class="csp-section">
        <h2>Available Tools</h2>
        <table class="csp-table">
            <thead><tr><th>Tool</th><th>Purpose</th></tr></thead>
            <tbody>
                <tr><td><code>getContext</code></td><td>Get server documentation, workflow patterns, API structure</td></tr>
                <tr><td><code>choose_lang</code></td><td>Set language (English or Latvian)</td></tr>
                <tr><td><code>get_topics</code></td><td>Browse statistical domains (Population, Economy, etc.)</td></tr>
                <tr><td><code>get_topic_content</code></td><td>Get sub-categories within a domain</td></tr>
                <tr><td><code>get_titles</code></td><td>List data tables in a category</td></tr>
                <tr><td><code>get_tables_by_keyword</code></td><td>Search tables by keyword</td></tr>
                <tr><td><code>filter_by_code</code></td><td>Filter tables by code pattern</td></tr>
                <tr><td><code>get_query_values</code></td><td>Get dimensions, filter values, and subtable info for a report</td></tr>
                <tr><td><code>get_csp_data</code></td><td>Retrieve statistical data with filters and subtable selection</td></tr>
                <tr><td><code>construct_csp_link</code></td><td>Generate URL to view report on data.stat.gov.lv</td></tr>
            </tbody>
        </table>
