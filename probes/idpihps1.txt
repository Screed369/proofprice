    function categoryChanged(e) {
        if ($("#cboCategory").dxSelectBox("instance").option('value') != 0) {
            $("#cboCommodity").dxSelectBox("instance").option('disabled', false)
            $("#cboCommodity").dxSelectBox("instance").getDataSource().reload();
            $("#cboCommodity").dxSelectBox("instance").option('value', 0);
        } else {
            $("#cboCommodity").dxSelectBox("instance").option('disabled', true)
            $("#cboCommodity").dxSelectBox("instance").getDataSource().reload();
            $("#cboCommodity").dxSelectBox("instance").option('value', null);
        }
    }
