<script>
document.addEventListener("DOMContentLoaded", function () {
  const input = document.getElementById("siteSearchInput");
  const button = document.getElementById("siteSearchBtn");
  const resultsBox = document.getElementById("searchResults");
  const didYouMean = document.getElementById("didYouMean");
  const micBtn = document.getElementById("micBtn");
  const container = document.querySelector(".dmpr-search");

  /* ----------------------------
     Build lightweight page index
     ---------------------------- */
  let index = [];
  document.querySelectorAll("a").forEach(a => {
      const text = (a.textContent || "").trim();
      const href = a.getAttribute("href");
      if (text.length > 2 && href && !href.startsWith("#") && !href.startsWith("javascript:") && !href.startsWith("mailto:")) {
          index.push({ title: text, url: a.href });
      }
  });
  // unique by title
  index = index.filter((v, i, a) => a.findIndex(t => t.title === v.title) === i);

  /* ----------------------------
     Utility: Levenshtein (for "Did you mean?")
     ---------------------------- */
  function levenshtein(a, b) {
    if (!a.length) return b.length;
    if (!b.length) return a.length;
    const matrix = [];
    for (let i = 0; i <= b.length; i++) matrix[i] = [i];
    for (let j = 0; j <= a.length; j++) matrix[0][j] = j;
    for (let i = 1; i <= b.length; i++) {
      for (let j = 1; j <= a.length; j++) {
        if (b.charAt(i-1) === a.charAt(j-1)) matrix[i][j] = matrix[i-1][j-1];
        else matrix[i][j] = Math.min(matrix[i-1][j] + 1, matrix[i][j-1] + 1, matrix[i-1][j-1] + 1);
      }
    }
    return matrix[b.length][a.length];
  }

  /* ----------------------------
     Autocomplete + Did you mean logic
     ---------------------------- */
  let debounceTimer = null;
  input.addEventListener("input", () => {
    container.classList.add("focused");
    clearTimeout(debounceTimer);
    debounceTimer = setTimeout(() => {
      const q = input.value.toLowerCase().trim();
      resultsBox.innerHTML = "";
      didYouMean.style.display = "none";
      didYouMean.innerHTML = "";

      if (q.length < 2) {
        resultsBox.style.display = "none";
        return;
      }

      // find matches in title
      const matches = index.filter(item => item.title.toLowerCase().includes(q)).slice(0, 8);

      if (matches.length > 0) {
        resultsBox.style.display = "block";
        matches.forEach(m => {
          const div = document.createElement("div");
          div.className = "result";
          // highlight matched portion
          const regex = new RegExp("(" + q.replace(/[.*+?^${}()|[\]\\]/g, "\\$&") + ")", "gi");
          div.innerHTML = m.title.replace(regex, "<span class='search-highlight'>$1</span>");
          div.onclick = () => window.location.href = m.url;
          resultsBox.appendChild(div);
        });
      } else {
        // No matches -> try "Did you mean?" using fuzzy match
        // compute distance for titles and pick best candidate
        let candidates = index.map(it => ({ title: it.title, url: it.url, dist: levenshtein(q, it.title.toLowerCase()) }));
        candidates.sort((a,b) => a.dist - b.dist);
        const top = candidates.slice(0,3).filter(c => c.dist <= Math.max(2, Math.floor(q.length * 0.4)));
        if (top.length) {
          didYouMean.style.display = "block";
          didYouMean.innerHTML = `<span class="label">Did you mean?</span> ${top.map(t => `<a href="${t.url}">${t.title}</a>`).join(' / ')}`;
        } else {
          resultsBox.style.display = "block";
          const div = document.createElement("div");
          div.className = "result";
          div.textContent = "No matches found. Press Enter to search site.";
          resultsBox.appendChild(div);
        }
      }
    }, 120);
  });

  // close when clicking outside
  document.addEventListener("click", (e) => {
    if (!container.contains(e.target)) {
      resultsBox.style.display = "none";
      didYouMean.style.display = "none";
      container.classList.remove("focused");
    }
  });

  // keyboard: Enter performs site search
  input.addEventListener("keydown", (e) => {
  if (e.key === "Enter") {
    e.preventDefault();

    if (resultsBox.style.display === "block" &&
        resultsBox.textContent.includes("No matches found")) {

      resultsBox.innerHTML = `
        <div class="result">
          No results found on this site. Please try different keywords.
        </div>
      `;
      return;
    }

    performSearch();
  }
});


  /* ----------------------------
     Search performer (DNN-compatible)
     ---------------------------- */
    function performSearch() {
  const term = input.value.trim();
  if (!term) return;

  // Detect virtual directory automatically
  const basePath = window.location.pathname.split('/')[1];
  const root = basePath ? '/' + basePath : '';

  // DNN-friendly URL (SEO format)
  const url = root + "/Search-Results/Search/" + encodeURIComponent(term);

  window.location.href = url;
}
  button.addEventListener("click", performSearch);

  /* ----------------------------
     Voice search (Web Speech API)
     ---------------------------- */
  let recognition, recognizing = false;
  if ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window) {
    const SpeechRec = window.SpeechRecognition || window.webkitSpeechRecognition;
    recognition = new SpeechRec();
    recognition.lang = 'en-US';
    recognition.interimResults = true;
    recognition.maxAlternatives = 1;

    recognition.onstart = () => {
      recognizing = true;
      micBtn.classList.add('recording');
      micBtn.title = "Listening… click to stop";
      container.classList.add("focused");
    };

    recognition.onerror = (event) => {
      console.warn("Speech recognition error", event);
      recognizing = false;
      micBtn.classList.remove('recording');
      micBtn.title = "Voice search";
    };

    recognition.onend = () => {
      recognizing = false;
      micBtn.classList.remove('recording');
      micBtn.title = "Voice search";
    };

    recognition.onresult = (event) => {
      let interim = '';
      let final = '';
      for (let i = 0; i < event.results.length; ++i) {
        if (event.results[i].isFinal) final += event.results[i][0].transcript;
        else interim += event.results[i][0].transcript;
      }
      input.value = final || interim;
      // trigger input handler to show suggestions
      input.dispatchEvent(new Event('input'));
      // if final result found, auto-search after short delay
      if (final) {
        setTimeout(() => {
          performSearch();
        }, 450);
      }
    };
  } else {
    // hide mic button if not supported
    micBtn.style.display = 'none';
  }

  micBtn.addEventListener('click', (e) => {
    e.preventDefault();
    if (!recognition) return;
    if (recognizing) {
      recognition.stop();
    } else {
      recognition.start();
    }
  });

  /* ----------------------------
     Accessibility: focus styles
     ---------------------------- */
  input.addEventListener('focus', () => container.classList.add('focused'));
  input.addEventListener('blur', () => { /* keep until clicking outside */ });

});
