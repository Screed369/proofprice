var baseUrl ="https://plvr.land.moi.gov.tw/";  
 
var m = new Map();
	 
$(".advDownloadClass").hide();
function chgDownloadType(obj){
	downLoadType = obj.value;
	if (downLoadType == 2){
		$(".advDownloadClass").show();
	} else{ 
		$(".advDownloadClass").hide();
	}
}

function preDownload(){
	if (downLoadType==2){ 
		startOpenData();
	}else{ 
		var season = $("#historySeason_id option:selected").val();
		var fileFormat = $("#fileFormatId").val();
		var url = "/DownloadSeason?season="+season+"&type=zip&fileName=lvr_land"+fileFormat+".zip";
		$("#download-button").attr("href",url);
		var href = baseUrl + $('#download-button').attr('href');
		$("#download-button").click();
		window.location.href = href;   
	}
}

function init(){
	$("#totalMsg").html("");
	$("#nowMsg").html("");
	$("#mainMsg").attr("style","display:none;");
}

function startOpenData(){
	init();
	fs = new zip.fs.FS();
	
	var fileFormat = $("#fileFormatId").val();
	var season = $("#historySeason_id option:selected").val();
	var cnt = $(".checkBoxGrp" ).filter(':checked').length; 
	if (cnt < 1){ 
		alert("請選擇所要下載的區域及交易類別");
		$("#div_body").unmask();
		return false;
	}else{

		if (cnt > 3 ){ 
			alert("您選擇了檔案數共:" + cnt + "個，已超過設定限制3個，可以選「全國資料」再自行篩選");
			$("#div_body").unmask();
			return false;
		}
		var fOpt = {
				"msg" : "您選擇了檔案數共:" + cnt + "個 ，下載中，請稍候...",
				"overlay" : true,
				"zindex" : 99999
		};
		qt.util.showLoading(fOpt);
//		$("#div_body").mask("您選擇了檔案數共:" + cnt + "個 ，下載中，請稍候...");
	} 
	fs.root.addHttpContent(season, 
			baseUrl +"/DownloadSeason?season="+season);
// 	fs.root.addHttpContent("readme", 
// 			baseUrl +"/DownloadSeason?season="+season+"&fileName="+"readme");

	fs.root.addHttpContent("schema-land.csv", 
			baseUrl +"/DownloadSeason?season="+season+"&fileName="+"schema-land.csv");

	fs.root.addHttpContent("schema-main.csv", 
			baseUrl +"/DownloadSeason?season="+season+"&fileName="+"schema-main.csv");

	fs.root.addHttpContent("schema-park.csv", 
			baseUrl +"/DownloadSeason?season="+season+"&fileName="+"schema-park.csv");

	fs.root.addHttpContent("schema-build.csv", 
			baseUrl +"/DownloadSeason?season="+season+"&fileName="+"schema-build.csv");

	fs.root.addHttpContent("manifest.csv", 
			baseUrl +"/DownloadSeason?season="+season+"&fileName="+"manifest.csv");
 


	var processCnt= 0; 
	$( ".checkBoxGrp" ).each(function( index ) {
		 var writeFileName = $(this).val()+"."+fileFormat;
		 var write_parkFileName = $(this).val()+"_park."+fileFormat;
		 var write_landFileName = $(this).val()+"_land."+fileFormat;
		 var write_buildFileName = $(this).val()+"_build."+fileFormat;
		 if($(this).is(':checked')){ 
			processCnt++;
			fs.root.addHttpContent(writeFileName,  
					baseUrl +"/DownloadSeason?season="+season+"&fileName="+writeFileName);  
			fs.root.addHttpContent(write_parkFileName,  
					baseUrl +"/DownloadSeason?season="+season+"&fileName="+write_parkFileName);  
			fs.root.addHttpContent(write_landFileName,  
					baseUrl +"/DownloadSeason?season="+season+"&fileName="+write_landFileName);  
			fs.root.addHttpContent(write_buildFileName,  
					baseUrl +"/DownloadSeason?season="+season+"&fileName="+write_buildFileName);  
// 			console.log(" fullurl:" + baseUrl +"/Download?fileName="+writeFileName);  
		 }  
	});   
	startDownload();  
}
